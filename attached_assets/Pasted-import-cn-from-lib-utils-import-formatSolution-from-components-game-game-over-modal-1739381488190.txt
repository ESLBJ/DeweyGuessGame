import { cn } from "@/lib/utils";
import { formatSolution } from "@/components/game/game-over-modal";
interface GridProps {
  guesses: string[];
  solution: string;
  currentRow: number;
  currentGuess: string;
  setCurrentGuess: (guess: string) => void;
  onSubmitGuess: () => void;
}
export function Grid({ guesses, solution, currentRow, currentGuess, setCurrentGuess, onSubmitGuess }: GridProps) {
  const rows = Array(6).fill("");
  // Format the solution here
  const formattedSolution = typeof solution === 'string' ? formatSolution(solution) : solution;
  const handleKeyDown = (event: KeyboardEvent) => {
    if (event.key === 'Backspace') {
      setCurrentGuess(currentGuess.slice(0, -1));
    } else if (event.key === 'Enter') {
      onSubmitGuess();
    } else if (event.key === 'Delete') {
      setCurrentGuess('');
    } else if (event.key.length === 1 && event.key >= '0' && event.key <= '9') {
      setCurrentGuess(currentGuess + event.key);
    }
  };
  return (
    <div className="flex justify-center items-center">
    <div className="inline-grid gap-[2px] mx-auto" onKeyDown={handleKeyDown}>
      {rows.map((_, i) => {
        const isCurrentRow = i === currentRow;
        const guess = isCurrentRow ? currentGuess : guesses[i] || "";
        return (
          <div key={i} className="grid grid-cols-6 gap-[2px] relative">
            {formattedSolution.length > 3 && (
              <div className="absolute left-[calc(50%-2px)] bottom-0 -translate-y-[calc(-20%-2px)] items-start w-1 h-1 bg-primary rounded-full" />
            )}
            {Array(6).fill("").map((_, j) => {
              const digit = guess[j];
              const solutionDigit = formattedSolution[j]; // Use formattedSolution
              let bgColor = "bg-background";
              let textColor = "text-foreground";
              let borderColor = "border-muted";
              if (digit && !isCurrentRow) {
                if (digit === solutionDigit) {
                  bgColor = "bg-green-500";
                  textColor = "text-primary-foreground";
                  borderColor = "border-green-500";
                } else if (formattedSolution.includes(digit)) { // Use formattedSolution
                  const guessCount = guess.slice(0, j + 1).split('').filter(d => d === digit).length;
                  const solutionCount = formattedSolution.split('').filter(d => d === digit).length; // Use formattedSolution
                  if (guessCount <= solutionCount) {
                    bgColor = "bg-yellow-500";
                    textColor = "text-primary-foreground";
                    borderColor = "border-yellow-500";
                  } else {
                    bgColor = "bg-muted";
                    textColor = "text-foreground";
                  }
                } else {
                  bgColor = "bg-muted";
                  textColor = "text-foreground";
                }
              }
              return (
                <div
                  key={j}
                  className={cn(
                    "w-12 h-12 border-2 flex items-center justify-center text-lg font-bold rounded-sm transition-colors",
                    bgColor,
                    textColor,
                    borderColor,
                    isCurrentRow && digit && "border-primary",
                    isCurrentRow && !digit && "border-muted",
                    !isCurrentRow && !digit && "border-muted"
                  )}
                  onKeyDown={handleKeyDown}
                >
                  {digit}
                </div>
              );
            })}
          </div>
        );
      })}
      </div>
    </div>
  );
}
